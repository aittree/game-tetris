### 2.3 对局（`game.py`）

| 用例 | 场景 | 期望 |
| --- | --- | --- |
| UT-G-01 | 无操作推进 800 ms（等级 1） | 方块恰好下落 1 格 |
| UT-G-02 | 连续推进 4000 ms | 下落 5 格，不出现「一次掉多格」 |
| UT-G-03 | 贴底后推进 500 ms | 锁定并生成下一块 |
| UT-G-04 | 贴底后移动一次再推进 500 ms | 锁定时间从移动后重新计算 |
| UT-G-05 | 贴底后连续移动 200 次 | 锁定延迟重置不超过 15 次，最终锁定 |
| UT-G-06 | 填 1 行后硬降 | `lines == 1`，得分含 `100 × level` |
| UT-G-07 | 同列缺口填满 4 行 | `lines == 4`，得分含 `800 × level`，触发 `tetris` 事件 |
| UT-G-08 | 人为设 `lines=9` 再消 1 行 | `level == 2`，触发 `levelup` 事件 |
| UT-G-09 | 暂存一次后 | `hold_type` 为原方块，`hold_used=True` |
| UT-G-10 | 未落定再次暂存 | 无效果，方块不变 |
| UT-G-11 | 锁定后暂存 | `hold_used` 复位，可再次暂存 |
| UT-G-12 | 生成位置被占 | 立即 `GAME_OVER`，`piece is None` |
| UT-G-13 | 方块整体锁定在隐藏行 | `GAME_OVER`（锁出） |

```python
import random
import pytest

from src.pieces import (SHAPES, PIECE_TYPES, ROTATIONS, SevenBag,
                        Piece, kicks_for, _rotate_cw)


def test_all_rotations_have_four_cells():
    for name in PIECE_TYPES:
        for rot in range(4):
            cells = sum(sum(row) for row in ROTATIONS[name][rot])
            assert cells == 4, f"{name} rot={rot}"


def test_o_piece_never_changes():
    assert len(set(ROTATIONS["O"])) == 1


def test_t_piece_states_differ():
    assert len(set(ROTATIONS["T"])) == 4


def test_four_rotations_return_to_start():
    for name in ("I", "T", "S"):
        m = SHAPES[name]
        r = m
        for _ in range(4):
            r = _rotate_cw(r)
        assert r == m


@pytest.mark.parametrize("name,count", [(n, 5) for n in ("I", "J", "L", "S", "T", "Z")])
def test_kick_tables_have_five_offsets(name, count):
    for f in range(4):
        offsets = kicks_for(name, f, (f + 1) % 4)
        assert len(offsets) == count
        assert offsets[0] == (0, 0)


def test_o_piece_has_single_kick():
    assert kicks_for("O", 0, 1) == ((0, 0),)


def test_bag_fairness_over_70_pieces():
    bag = SevenBag(random.Random(1))
    types = [bag.next() for _ in range(70)]
    for name in PIECE_TYPES:
        assert types.count(name) == 10


def test_bag_has_no_duplicate_in_one_bag():
    bag = SevenBag(random.Random(7))
    first = [bag.next() for _ in range(7)]
    assert sorted(first) == sorted(PIECE_TYPES)


def test_bag_reproducible_with_seed():
    bag1 = SevenBag(random.Random(42))
    seq1 = [bag1.next() for _ in range(14)]
    bag2 = SevenBag(random.Random(42))
    assert seq1 == [bag2.next() for _ in range(14)]


def test_piece_cells_shift():
    p = Piece("T", x=3, y=5)
    up = p.cells()
    down = p.cells(dy=-1)
    assert [(r + 1, c) for r, c in up] == down
```

`tests/test_board.py`：

```python
import pytest
from src.board import Board
from src.pieces import Piece

EMPTY_ROWS = ["." * 10] * 22


def make(rows):
    b = Board()
    b.load_from(rows)
    return b


def test_collides_inside_empty_board():
    b = Board()
    assert b.collides([(5, 5)]) is False


@pytest.mark.parametrize("cell", [(5, -1), (5, 10), (22, 5)])
def test_collides_out_of_bounds(cell):
    assert Board().collides([cell]) is True


def test_collides_above_field_allowed():
    assert Board().collides([(-1, 5)]) is False


def test_lock_and_full_rows():
    b = Board()
    b.load_from(["IIIIIIIII."] + EMPTY_ROWS[1:])
    b.lock(Piece("O", x=8, y=0))            # O 占 (0,8)(0,9)(1,8)(1,9)
    assert b.grid[0][9] == "O"
    assert b.full_rows() == [0]             # 第 0 行被 O 补满


def test_clear_two_rows_shifts_board():
    rows = ["IIIIIIIIII", "IIIIIIIIII", "JJJ......."] + EMPTY_ROWS[3:]
    b = make(rows)
    b.clear_rows([0, 1])
    assert b.grid[0] == [0] * 10
    assert b.grid[1] == [0] * 10
    assert b.grid[2][:3] == ["J", "J", "J"]


def test_visible_count_ignores_hidden_rows():
    b = make(["IIIIIIIIII", "IIIIIIIIII"] + EMPTY_ROWS[2:])
    assert b.visible_count() == 0
```

`tests/test_game.py`：

```python
import pytest
from src.board import Board
from src.game import Game, GameState, Action
from src.pieces import Piece


def empty_rows():
    return ["." * 10] * 22


@pytest.fixture
def game():
    g = Game(best=0)
    return g


def test_gravity_one_cell_in_800ms(game):
    game.piece = Piece("O", x=4, y=2)
    start = game.piece.y
    game.update(800)
    assert game.piece.y == start + 1


def test_gravity_two_cells_in_1600ms(game):
    game.piece = Piece("O", x=4, y=2)
    start = game.piece.y
    game.update(1600)
    assert game.piece.y - start == 2       # 逐格推进，不会一次掉到地面


def test_lock_after_500ms(game):
    game.piece = Piece("O", x=4, y=20)
    game.update(1)
    game.update(500)
    assert game.piece is not None          # 已锁定并生成下一块（新对象）
    assert game.board.visible_count() == 4


def test_lock_reset_limit(game):
    game.piece = Piece("O", x=4, y=20)      # 占据 20、21 两行，已贴底
    game.update(1)
    for _ in range(20):
        game.handle(Action.MOVE_LEFT)
        game.handle(Action.MOVE_RIGHT)
        game.update(20)
    game.update(600)
    assert game.board.visible_count() > 0   # 达到重置上限后仍然锁定成功


def test_single_line_score(game):
    game.board.load_from(["." * 10] * 21 + ["IIIIIIIII."])   # 仅第 21 行缺 1 格
    game.piece = Piece("I", rot=1, x=7, y=0)                # 竖 I 的填充列 = x+2 = 9
    game.handle(Action.HARD_DROP)
    assert game.lines == 1
    assert game.score >= 100


def test_tetris_score(game):
    game.board.load_from(["." * 10] * 18 + ["IIIIIIIII."] * 4)
    game.piece = Piece("I", rot=1, x=7, y=0)
    game.handle(Action.HARD_DROP)
    assert game.lines == 4
    assert game.score >= 800


def test_hold_only_once_per_piece(game):
    first = game.piece.type
    game.handle(Action.HOLD)
    assert game.hold_type == first and game.hold_used is True
    second = game.piece.type
    game.handle(Action.HOLD)
    assert game.piece.type == second       # 第二次无效


def test_block_out_ends_game():
    g = Game()
    g.board.load_from(["IIIIIIIIII"] + ["." * 10] * 21)
    g.piece = Piece("O", x=4, y=0)
    g.spawn("O")                           # 生成位置已被占用
    assert g.state is GameState.GAME_OVER
```

import json
import pytest
from src import storage


@pytest.fixture(autouse=True)
def temp_save(tmp_path, monkeypatch):

def test_missing_file_returns_defaults():

    storage.SAVE_FILE.write_text(json.dumps({"best": 1234}), encoding="utf-8")

def test_broken_json_falls_back():
    storage.SAVE_FILE.write_text("{not json", encoding="utf-8")
    assert storage.load_profile().best == 0


def test_roundtrip():
    p = storage.load_profile()
    p.best = 900
    p.settings.music = 0
    storage.save_profile(p)
    assert storage.load_profile().settings.music == 0
